cp_setjmp:
  asm
  mov
    r0
    r12
  mov
    sp
    r3
  str.post
    4
    r12
    r3
  str.post
    4
    r12
    lr
  stmia.wb
    0xff0
    r12
  movi
    0
    0
    r0
  bx
    lr
cp_longjmp:
  asm
  mov
    r0
    r12
  ldr.post
    4
    r12
    r3
  ldr.post
    4
    r12
    r4
  mov
    r3
    sp
  mov
    r4
    lr
  ldmia.wb
    0xff0
    r12
  movi
    0
    1
    r0
  bx
    lr
